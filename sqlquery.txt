CREATE TABLE USER (
    USER_Id INT AUTO_INCREMENT PRIMARY KEY,
    USER_FName VARCHAR(50),
    USER_LName VARCHAR(50),
    USER_Password VARCHAR(255),
    USER_Age INT,
    USER_Email VARCHAR(100),
    USER_Gender VARCHAR(1),
    USER_Occupation VARCHAR(100),
    USER_Establishment TEXT,
    USER_Points INT DEFAULT 0,
    USER_LoginStatus INT DEFAULT 0
);

CREATE TABLE STORE (
    STORE_Id INT AUTO_INCREMENT PRIMARY KEY,
    STORE_NumOfProducts INT,
    USER_Id INT,
    FOREIGN KEY (USER_Id) REFERENCES USER(USER_Id)
);

CREATE TABLE PRODUCT (
    PROD_Id INT AUTO_INCREMENT PRIMARY KEY,
    PROD_Name VARCHAR(100) NOT NULL,
    PROD_Category VARCHAR(100),
    PROD_Amount INT,
    STORE_Id INT,
    PROD_Photo LONGBLOB,
    FOREIGN KEY (STORE_Id) REFERENCES STORE(STORE_Id)
);

UPDATE store SET STORE_NumOfProducts = (
    SELECT COUNT(*) FROM product
    WHERE product.STORE_Id = store.STORE_Id
);

CREATE TABLE CALENDAR (
    CAL_Id INT AUTO_INCREMENT PRIMARY KEY,
    CAL_NumOfTask INT,
    USER_Id INT,
    FOREIGN KEY (USER_Id) REFERENCES USER(USER_Id)
);

CREATE TABLE TASK (
    TASK_Id INT AUTO_INCREMENT PRIMARY KEY,
    TASK_Name VARCHAR(255),
    TASK_Description TEXT NOT NULL,
    TASK_PriorityClass VARCHAR(50),
    TASK_Status VARCHAR(50),
    TASK_DateStart DATE,
    TASK_DateFinish DATE,
    TASK_Progress INT,
    CAL_Id INT,
    USER_Id INT,
    FOREIGN KEY (CAL_Id) REFERENCES CALENDAR(CAL_Id),
    FOREIGN KEY (USER_Id) REFERENCES USER(USER_Id)
);

CREATE TABLE REMINDER (
    REMIND_Id INT AUTO_INCREMENT PRIMARY KEY,
    REMIND_Date DATE NOT NULL,
    REMIND_Text VARCHAR (255),
    TASK_Id INT
);


CREATE TABLE REWARD (
    REWARD_Id INT AUTO_INCREMENT PRIMARY KEY,
    REWARD_Points INT NOT NULL,
    TASK_Id INT,
    USER_Id INT,
    FOREIGN KEY (TASK_Id) REFERENCES TASK(TASK_Id),
    FOREIGN KEY (USER_Id) REFERENCES USER(USER_Id)
);

CREATE TABLE purchases (
    id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NOT NULL,
    prod_id INT NOT NULL,
    purchased_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE KEY unique_user_pet (user_id, prod_id),
    FOREIGN KEY (user_id) REFERENCES user(USER_Id),
    FOREIGN KEY (prod_id) REFERENCES product(PROD_Id)
);
